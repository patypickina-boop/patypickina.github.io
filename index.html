async function searchCEP() { const cep = document.getElementById('cep-input').value.replace(/\D/g, ''); if (cep.length !== 8) return showStatus("CEP inválido. Digite 8 números.", "error"); showStatus("Localizando endereço...", "loading"); try { // PASSO 1: BrasilAPI v2 (Busca coordenada direta do CEP) const r1 = await fetch(`https://brasilapi.com.br/api/cep/v2/${cep}`); const d1 = await r1.json(); if (d1.location && d1.location.coordinates) { const lat = parseFloat(d1.location.coordinates.latitude); const lon = parseFloat(d1.location.coordinates.longitude); // Valida se o ponto está no DF (Lat: -16.1 a -15.4 | Lon: -48.3 a -47.2) if (lat < -15.4 && lat > -16.1) { return updateOrigin([lat, lon], `CEP ${cep}`); } } // PASSO 2: Fallback ViaCEP + Nominatim Estruturado const r2 = await fetch(`https://viacep.com.br/ws/${cep}/json/`); const d2 = await r2.json(); if (d2.erro) throw new Error("CEP não encontrado."); // Limpeza para Brasília: "SHIS QI 23 Conjunto 1" -> "SHIS QI 23" const cleanAddr = d2.logradouro.split('Conjunto')[0].split('Lote')[0].trim(); const query = `${cleanAddr}, ${d2.bairro}, Distrito Federal, Brasil`; const r3 = await fetch(`https://nominatim.openstreetmap.org/search?q=${encodeURIComponent(query)}&format=json&limit=1&viewbox=-48.28,-15.45,-47.30,-16.05&bounded=1`); const d3 = await r3.json(); if (d3.length > 0) { updateOrigin([parseFloat(d3[0].lat), parseFloat(d3[0].lon)], `CEP ${cep} (${d2.bairro})`); } else { showStatus("CEP ok, mas localização exata não encontrada. Use o GPS.", "error"); } } catch (e) { showStatus("Erro na busca. Verifique o CEP.", "error"); } }
