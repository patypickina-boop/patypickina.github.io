async function searchCEP() { const cep = document.getElementById('cep-input').value.replace(/\D/g, ''); if (cep.length !== 8) return showStatus("CEP inválido. Digite 8 números.", "error"); showStatus("Localizando endereço...", "loading"); try { // PASSO 1: ViaCEP -> endereço por extenso -> Nominatim (precisão de RUA). // As coordenadas postais da BrasilAPI no DF são imprecisas e ficam // como ÚLTIMO recurso (eram a causa do pino cair no lugar errado). const r2 = await fetch(`https://viacep.com.br/ws/${cep}/json/`); const d2 = await r2.json(); if (d2.erro) throw new Error("CEP não encontrado."); const tentativas = [ `${d2.logradouro}, ${d2.bairro}, ${d2.localidade}, ${d2.uf}, Brasil`, `${d2.logradouro}, ${d2.localidade}, ${d2.uf}, Brasil`, `${d2.bairro}, ${d2.localidade}, ${d2.uf}, Brasil` ].filter(q => !q.startsWith(', ')); for (const query of tentativas) { const r3 = await fetch(`https://nominatim.openstreetmap.org/search?q=${encodeURIComponent(query)}&format=json&limit=1&viewbox=-48.28,-15.45,-47.30,-16.05&bounded=1`); const d3 = await r3.json(); if (d3.length > 0) { updateOrigin([parseFloat(d3[0].lat), parseFloat(d3[0].lon)], `CEP ${cep} — ${d2.logradouro || d2.bairro}`); return; } } // PASSO 2 (último recurso): coordenadas da BrasilAPI, agora com // validação de latitude E longitude dentro do DF const r1 = await fetch(`https://brasilapi.com.br/api/cep/v2/${cep}`); if (r1.ok) { const d1 = await r1.json(); if (d1.location && d1.location.coordinates) { const lat = parseFloat(d1.location.coordinates.latitude); const lon = parseFloat(d1.location.coordinates.longitude); if (lat < -15.4 && lat > -16.1 && lon < -47.2 && lon > -48.3) { return updateOrigin([lat, lon], `CEP ${cep} (aproximado)`); } } } showStatus("CEP ok, mas localização exata não encontrada. Use o GPS.", "error"); } catch (e) { showStatus("Erro na busca. Verifique o CEP.", "error"); } }
